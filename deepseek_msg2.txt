【一、navigate 的等待】
async def _wait_document(self, before, loader_id, timeout):
    deadline = loop.time() + timeout
    settled = 0
    while loop.time() < deadline:
        await asyncio.sleep(0.04)
        state = await self._nav_state()   # {href, ready, t0, hist}
        moved = (state["href"] != before["href"]
                 or state["hist"] != before["hist"]
                 or state["t0"] != before["t0"])
        if moved and state["ready"] != "loading":
            settled += 1
            if settled >= 2 or state["ready"] == "complete":
                return
        else:
            settled = 0
    raise AttachError("Navigation did not settle within ...")
